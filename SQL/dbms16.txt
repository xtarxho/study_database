--employees table 복사
create table empl AS
select * from employees;

1) update
update는 empl로 접근해서 set으로 들어간다
update empl set salary = 10000;
commit;

--근속년수가 20년 이상인 사원들만 월급을 10000 인상
update empl 
set salary = salary + 10000
where months_between(sysdate, hire_date) >= 12 * 20;

--100번 사원의 급여를 10% 인상
update empl
set salary = salary + salary * 0.1
where employee_id = 100;

--IT_PROG직종의 사원들 급여를 5% 인상
update empl
set salary = salary + salary * 0.05
where job_id = 'IT_PROG';

--30번 부서의 사원들의 커미션을 0.1로 변경
update empl
set commission_pct = 0.1
where department_id = 30;

--모든 직원들의 월급을 10000으로 수정
update empl
set salary = 10000;

--empl 안에 있는 데이터를 삭제
delete from empl;

--empl 안에 데이터 넣기
insert into empl
select * from employees;
commit;

--전체 직원의 평균 월급보다 적게받는(미만) 직원들의 월급을 500 인상
update empl 
set salary = salary + 500
where salary < (select avg(salary) from empl);

--30번 부서의 평균 급여보다 적게 받는 사원들의 월급을 300 인상
update empl
set salary = salary + 300
where salary < (select avg(salary) from empl where department_id = 30);

--dept table 복사
create table dept AS
select * from departments;

--부서명이 IT인 직원들의 월급을 10% 인상
update empl
set salary = salary + salary * 0.1
where department_id = (select department_id from dept where department_name = 'IT');

--부서이름이 IT인 사원중 급여가 6000미만인 직원들의 급여를 1000 인상
update empl
set salary = salary + 1000
where department_id = (select department_id from dept where department_name = 'IT') 
and salary < 6000;

--부서이름이 Sales이거나 Marketing인 직원들의 이메일을 x로 변경
update empl
set email = 'x'
where department_id in(select department_id from dept where department_name in ('Sales', 'Marketing'));

--shipping부서 직원들의 휴대폰 번호를 x로 수정
update empl
set phone_number = 'x'
where department_id = (select department_id from dept where department_name = 'Shipping');

2) delete
delete from dept
where department_name = 'Shipping';
commit;

--30번 부서의 모든 직원을 해고
delete from dept
where department_id = 30;
commit;

--보너스를 받지 않는 직원들을 삭제
delete from empl
where commission_pct = null;

--월급이 7000이상 10000이하의 직원들을 삭제
delete from empl
where salary >= 7000 and salary <= 10000;

drop table empl;

delete from empl
where salary < (select avg(salary) from empl where department_id = 30);

--부서이름이 IT이거나 Sales인 직원들을 삭제
delete from empl
where department_id in (select department_id from dept where department_name in ('IT', 'Sales'));

--부서명이 s로 시작하는 부서를 삭제
delete from dept
where department_name like 'S%';

--부서명이 15글자 이하인 부서를 삭제
delete from dept
where length(department_name) <= 15;
