1) like연산자(유사검색)
사용자가 지정한 문자형태와 일치 또는 포함하고 있는 자원을 검색한다
% : 모든 값을 찾으라는 뜻
_ : 하나의 값을 찾으라는 뜻
예1) 'a%' : a로 시작하는 모든 값(자원)을 찾는다
예2) '%a' : a로 끝나는 모든 값(자원)을 찾는다
예3) '%a%' : 전체 내용 중에 a가 한글자라도 있으면 찾는다
예4) 'a____'(언더바 4개임) : a로 시작하는 값 중에 길이가 5인 자원을 찾는다

--이름이 D로 시작하는 사원들의 사번, 이름을 출력
select employee_id, first_name
from employees
where first_name LIKE 'D%';

--이름이 d로 끝나는 사원들의 사번, 이름, 성을 출력
select employee_id, first_name, last_name
from employees
where first_name like '%d';

--이름의 3번째 글자가 a인 사원들의 사번, 이름을 출력
select employee_id, first_name
from employees
where first_name like '__a%';

--이름이 6글자인 사원중에 4번째 글자가 a인 직원들의 정보를 사번, 이름 순으로 출력
select employee_id, first_name
from employees
where first_name like '___a__';

--이름의 시작이 s 끝이 n인 직원들의 정보를 사번, 이름, 직종순으로 출력
select employee_id, first_name, job_id
from employees
where first_name like 'S%n';

--직종에 SA, IT가 포함되어 있지 않은 사원들의 정보를 사번, 이름, 직종 순으로 출력
select employee_id, first_name, job_id
from employees
where job_id not like 'SA%' and job_id not like 'IT%';

--이름이 H로 시작하면서 6글자 이상인 사원들의 정보를 이름, 직종 순으로 출력
select first_name, job_id
from employees
where first_name like 'H_____%';

--이름에 소문자 o가 포함되어 있으면서 a로 끝나는 사원들의 사번, 이름을 출력
select employee_id, first_name
from employees
where first_name like '%o%a';

--전화번호가 011로 시작하는 사원들의 이름, 전화번호를 출력
select first_name, phone_number
from employees
where phone_number like '011%';

--전화번호가 6으로 시작해서 9로 끝나면서 중간에 2가 들어가는 사원들의 정보를 이름, 전화번호 순으로 조회
select first_name, phone_number
from employees
where phone_number like '6%2%9';

-- 전화번호에 44가 포함되어 있는 사원들의 이름, 전화번호를 조회
select first_name, phone_number
from employees
where phone_number like '%44%';

--입사년도에 상관없이 1월에 입사한 모든 사원들의 정보를 사번, 이름, 입사일 순으로 출력
select employee_id, first_name, hire_date
from employees
where hire_date like '01%';

--전화번호에 44가 포함되어 있지 않은 사원들의 정보를 이름, 전화번호 순으로 조회
select first_name, phone_number
from employees
where phone_number not like '%44%';

--44가 전화번호 중간에만 들어있는 사원들의 이름, 전화번호를 조회
select first_name, phone_number
from employees
where phone_number like '%44%' and phone_number not like '44%' and phone_number not like '%44';
