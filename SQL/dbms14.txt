--사원 이름이 Michael이고, 직종이 mk_man인 사원보다 월급을 많이 받은(초과) 사원들의 정보를 사번, 이름, 직종, 급여순으로 조회

--문제 접근1)
--이름이 마이클이고 직종이 mk_man인 사원의 급여 -> 13000
select salary
from employees 
where first_name = 'Michael' and job_id = 'MK_MAN';

--문제 접근 2)
--13000을 초과하는 급여를 받는 사원들의 정보
select employee_id, first_name, job_id, salary
from employees
where salary > 13000;

--위의 두 select문 합치기
select employee_id, first_name, job_id, salary
from employees
where salary > (select salary
from employees 
where first_name = 'Michael' and job_id = 'MK_MAN');

--회사에서 월급을 가장 많이받는 사원의 사번, 이름, 월급을 조회
select employee_id, first_name, salary
from employees
where salary = (select max(salary) from employees);

--150번 사원과 같은 월급을 받는 모든 사원들의 정보를 사번, 이름, 급여순으로 출력
select employee_id, first_name, salary
from employees
where salary = (select salary from employees where employee_id = 150);

--Bruce와 같은 부서에서 근무중인 사원들의 사번, 이름을 출력
select employee_id, first_name
from employees
where department_id = (select department_id from employees where first_name = 'Bruce');

--111번 사원과 같은 직종이면서 159번 사원의 급여를 초과하는 사원들의 정보를 사번, 이름, 직종, 입사일, 급여순으로 조회
select employee_id, first_name, job_id, hire_date, salary
from employees
where job_id = (select job_id from employees where employee_id = 111) 
and salary > (select salary from employees where employee_id = 159);

--직종별 급여의 평군을 출력 
평균 급여가 Bruce의 급여를 초과하는 경우만 출력
select job_id, avg(salary)
from employees
having avg(salary) > (select salary from employees where first_name = 'Bruce')
group by job_id;

---숙제---
유럽지역 사원들의 평균 급여보다 많이받는(초과) 사원들의 이름, 급여, 지역을 조회
select e.first_name, e.salary, r.region_name
from employees e
join departments d on e.department_id = d.department_id
join locations l on d.location_id = l.location_id
join countries c on l.country_id = c.country_id
join regions r on c.region_id = r.region_id
where e.salary > (
select avg(e2.salary) 
from employees e2
join departments d2 on e2.department_id = d2.department_id
join locations l2 on d2.location_id = l2.location_id
join countries c2 on l2.country_id = c2.country_id
join regions r2 on c2.region_id = r2.region_id
where r2.region_name = 'Europe'
);


--월급이 가장 적은 사원의 사번, 이름, 성, 월급을 조회
select employee_id, first_name, last_name, salary
from employees
where salary = (
select min(salary) from employees
);

--137번 사원보다 월급이 많고(이상) 149번 사원보다 월급이 적은(이하) 사원들의 이름, 월급을 조회
select first_name, salary
from employees
where salary 
between 
(select salary from employees where employee_id = 137) 
and 
(select salary from employees where employee_id = 149);

--직종이 sa_rep인 사원들의 최소급여보다 적으면서(미만) 직종이 sh_clerk은 아닌 사원들의 이름, 성, 직종, 급여를 조회
select first_name, last_name, job_id, salary
from employees
where salary < (
select min(salary) from employees where job_id = 'SA_REP'
) and job_id != 'SH_CLERK';

--100번 부서의 최소 월급보다 최소월급이 많은(초과) 다른 모든 부서의 부서번호와 최소급여를 조회
select department_id, min(salary) min
from employees
group by department_id
having min(salary) > (select min(salary) from employees where department_id = 100);

--성 중에 Bat을 포함하고 있는 사원과 같은 부서에서 근무하는 사원들의 모든 정보를 출력
select * from employees
where department_id = (select department_id from employees where last_name like 'Bat%');

--158번 사원과 같은 직종에서 근무하는 사원들 중 급여가 164번 사원보다 많은(초과) 사원들의 사번, 이름, 성, 직종, 급여를 조회
select employee_id, first_name, last_name, job_id, salary
from employees
where salary > (select salary from employees where employee_id = 164)
and job_id = (select job_id from employees where employee_id = 158);

--Matthew를 상사로 두고 있으면서 직종이 st_clerk인 사원들의 사번, 이름, 직종, 상사번호를 출력
select employee_id, first_name, job_id, manager_id
from employees
where manager_id = (select employee_id from employees where first_name = 'Matthew') and job_id = 'ST_CLERK';

--근무도시가 Seattle인 사람들의 평균 급여보다 많이받는(초과) 사원들의 사번, 이름, 급여를 조회하시오
select employee_id, first_name, salary
from employees
where salary > (select avg(salary) from employees e 
join departments d on e.department_id = d.department_id
join locations l on d.location_id = l.location_id
where l.city = 'Seattle'
);

--시애틀 근무자들의 평균급여보다 많이받는(초과) 사람들을 시애틀 근무자들 중에서만 추려서 사번, 이름, 급여, 도시 순으로 출력
select e.employee_id, e.first_name, e.salary, l.city
from employees e
join departments d on e.department_id = d.department_id
join locations l on d.location_id = d.location_id
where l.city = 'Seattle'
and e.salary > (select avg(e2.salary) from employees e2 
join departments d2 on e2.department_id = d2.department_id
join locations l2 on d2.location_id = l2.location_id
where l2.city = 'Seattle');

--각 부서별 최고 월급을 받는 사원의 부서번호, 이름, 급여를 부서번호 오름차순으로 정렬
select e.department_id, e.first_name, e.salary
from employees e, (
select department_id, max(salary) mx
from employees
group by department_id
) max_sal
where e.salary = max_sal.mx
order by e.department_id;

--페이징처리--
--사원테이블 한페이지에 10명씩 조회되도록 페이징처리
select * from
(
select rank() over(order by employee_id ) no, e.*
from employees e 
)
where no between 11 and 20;
