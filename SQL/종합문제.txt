1. 사원번호가 126인 사람의 이름, 입사일자, 부서번호를 출력
select first_name, hire_date, department_id
from employees
where employee_id = 126;

2. 부서번호가 10번인 직원의 부서번호, 사번, 이름, 급여를 출력
select department_id, employee_id, first_name, salary
from employees
where department_id = 10;

3. 사원테이블의 전체 사원의 평균 임금보다 많은(초과) 사원의 사원번호, 이름, 부서명, 입사일, 급여를 급여오름차순으로 출력
select e.employee_id, e.first_name, d.department_name, e.hire_date, e.salary
from employees e
join departments d on e.department_id = d.department_id
where salary > (select avg(salary) from employees e )
order by salary;

4. 30번 부서의 평균 급여보다 많이 받는(초과) 직원들 중에서 가장 적은 급여를 받는 직원들의 이름과 급여를 출력
select first_name, salary
from employees

5. employees 테이블에서 급여를 많이 받는 순서대로 조회했을 때 6번째부터 10번째까지 5명의 last_name, first_name, salary를 조회
select ranking, last_name, first_name, salary
from (
    select rank() over(order by salary desc) ranking,
           last_name,
           first_name,
           salary
    from employees
)
where ranking between 6 and 10;

6. 각 업무별로 급여의 총합을 구하고자 한다. 급여 총합이 가장 높은 업무부터 업무명과 급여총합을 조회하시오. 단, 급여총합이 30,000보다 큰(초과) 업무만 출력하시오.
select j.job_title, sum(e.salary)
from employees e
join jobs j on e.job_id = j.job_id
group by j.job_title
having sum(e.salary) > 30000
order by sum(e.salary) DESC;

7. 각 사원에 대해서 사번, 이름, 업무명, 부서명을 조회하시오. 단, 도시명이 'Seattle'인 지역의 부서에 근무하는 직원을 사원번호 오름차순으로 출력하시오.
select e.employee_id, e.first_name, j.job_title, d.department_name
from employees e
join departments d on e.department_id = d.department_id
join jobs j on e.job_id = j.job_id
join locations l on d.location_id = l.location_id
where l.city = 'Seattle'
order by e.employee_id;

8. 'Sales' 부서에 속한 직원의 이름, 급여, 부서이름을 조회하시오.
단, 급여는 100번 부서의 평균보다 적게 받는(미만) 직원 정보만 출력되어야 한다.
select e.first_name, e.salary, d.department_name
from employees e
join departments d on e.department_id = d.department_id
where d.department_name = 'Sales' 
and e.salary < (select avg(e.salary) from employees e where e.department_id = 100);

9. 사원테이블에서 각 직종별 인원과 월급의 평균을 구해서 월급 평균의 내림차순으로 정렬하시오. 단, 인원이 3명 이상인 직종만 출력하시오.
select job_id 직업, count(*) 인원, avg(salary) 평균급여
from employees
group by job_id
having count(*) >= 3
order by avg(salary) DESC;

10. Americas 지역 사원 평균 급여보다 많이 받는 Europe 지역 사원만 사번, 이름, 급여, 지역 순으로 출력하고 급여가 많은 순으로 정렬하시오.
select e.employee_id, e.first_name, e.salary, r.region_name
from employees e
join departments d on e.department_id = d.department_id
join locations l on d.location_id = l.location_id
join countries c on l.country_id = c.country_id
join regions r on c.region_id = r.region_id
where r.region_name = 'Europe'
and e.salary > (select avg(e.salary) from employees e 
join departments d on e.department_id = d.department_id
join locations l on d.location_id = l.location_id
join countries c on l.country_id = c.country_id
join regions r on c.region_id = r.region_id
where r.region_name = 'Americas')
order by e.salary DESC;

11. employees 테이블에서 last_name이 Kochhar의 급여보다 많은 사원의 사원번호, 성, 담당업무, 월급을 화면에 출력하시오.
select employee_id, last_name, job_id, salary
from employees
where salary > (select salary from employees where last_name = 'Kochhar');

12. employees 테이블에서 job_id가 'AD_PRES'인 사원의 부하직원들을 검색하여 이름, 직책, 관리자ID를 출력하시오.
select first_name, job_id, manager_id
from employees
where manager_id = (select employee_id from employees where job_id = 'AD_PRES' );

13. 사원 테이블의 모든 사원의 이름을 앞의 세글지만 표기하고 뒷부분은 *로 변경하고 사번, 이름, 가린이름 순으로 출력하시오.
select employee_id, first_name, rpad(substr(first_name, 1, 3), 5, '*')
from employees;
