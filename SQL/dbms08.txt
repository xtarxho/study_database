1. 그룹함수
-avg() : 평균을 구하는 함수
-count() : 반환된 행의 갯수를 구하는 함수(null값은 무시한다)
-max() : 최대값을 구하는 함수
-min() : 최소값을 구하는 함수
-sum() : 합을 구하는 함수
# 그룹함수와 일반컬럼은 함께 사용할 수 없다.
select count(*), salary from employees

--모든 사원의 수를 구하시오
select count(*) from employees;
select count(employee_id) from employees;

--부서에 속해있는 사원의 수
select count(department_id) from employees;

--보너스를 받고있는 사원의 수
select count(commission_pct) from employees;

--선임이 있는 사원의 수
select count(manager_id) from employees;

--누군가의 상사로 일하고 있는 사원의 수
select count(distinct manager_id) from employees;

--80번 부서에 속하는 사원들의 월급의 평균을 출력
select avg(salary) from employees
where department_id = 80;

--50번 부서 사원들의 최대월급, 최소월급을 조회
select max(salary), min(salary) from employees
where department_id = 50;

--50번 부서 사원들의 월급의 합을 출력
select sum(salary) from employees
where department_id = 50;

--it_prog의 수를 출력
select count(*) from employees
where job_id = 'IT_PROG';

--sa_rep직종의 사원들의 월급의 평균, 최고월급, 최저월급, 월급의 합을 출력
select avg(salary), max(salary), min(salary), sum(salary)
from employees
where job_id = 'SA_REP';

--보너스를 받지 않는 사원의 수 출력
select count(*) from employees
where commission_pct is null;

--월급이 5000이하인 사원들의 평균 월급을 조회
select avg(salary) from employees
where salary <= 5000;

--2005년 이후 입사자의 수를 조회
select count(*) from employees
where hire_date >= '01/01/2005';

--우리회사에 있는 부서 수를 조회
select count(distinct department_id) from employees;

* nvl() : null값을 원하는 데이터로 교체하는 함수
--보너스를 받지 않는 모든 사원들의 보너스를 0.1%로 조정
select commission_pct,
	nvl(commission_pct, 0.1)
from employees

--우리회사에 있는 부서 수를 조회(대장님 포함)
select count(distinct nvl(department_id, 0)) from employees;

* dual(가상테이블) : 함수나 계산식을 테이블의 생성 없이 수행하여 볼 수 있도록 하는 가상(임시)의 테이블

2. 숫자함수
-round() : 반올림을 해 주는 함수
select round(0.123),round(0.567) from dual;
select round(0.125678, 2) from dual;

--80번 부서의 월급의 평균을 소수점 첫째자리까지 반올림
select round(avg(salary), 1) from employees
where department_id = 80;

--it_prog인 사원들의 평균 월굽을 소수점 두자리에서 반올림
select round(avg(salary),2) from employees
where job_id = 'IT_PROG';

-trunc() : 잘라내는(버림) 함수
--이름이 A로 시작하는 사원들의 평균월급을 소수점 1자리에서 절삭해서 조회
select trunc(avg(salary), 1) from employees
where first_name like 'A%';

-power() : 제곱을 구하는 함수
select power(4, 2) from dual;

-mod() : 나머지를 구하는 함수
select mod(7, 4) from dual;

--짝수 사번을 가진 사원들의 사번, 이름, 급여를 출력
select employee_id, first_name, salary from employees
where mod(employee_id, 2) = 0;

3. 문자함수
-initcap() : 첫 문자를 대문자로 변환하는 함수
-공백이나 /를 구분자로 인식한다.
select initcap('good morning') from dual;
select initcap('good/morning') from dual;

-lower() : 대문자 -> 소문자
-upper() : 소문자 -> 대문자
select lower('Good Morning') from dual;
select upper('good morning') from dual;

---밑에 두문제 다시 풀어보기---
--michael로 조회했을 때, 이름이 Michael인 사원의 사번, 이름을 조회
select employee_id, first_name
from employees
where lower(first_name) = 'michael';

--이름의 어디든 m또는 M이 들어있는 
사원의 사번, 이름을 조회
select employee_id, first_name
from employees
where lower(first_name) like '%m%';

-lpad() : 왼쪽의 자릿수를 늘리는 함수
select lpad('good', 6, '#') from dual;
select lpad('good', 8, '안녕') from dual;

-rpad() : 오른쪽의 자릿수를 늘리는 함수
select rpad('good', 6, '@') from dual;

-ltrim() : 왼쪽의 문자를 지우는 함수
select ltrim('helloh', 'h') from dual;

-rtrim() : 오른쪽의 문자를 지우는 함수
select rtrim('helloh', 'h') from dual;

-substr() : 특정 문장을 잘라내는 함수
select substr('good morning michael', 8, 4) from dual;
select substr('good morning michael', 8) from dual;
select substr('good morning michael', -4) from dual; 
select substr('good morning michael', -7, 4) from dual;

--이름, 입사연도만 출력
select first_name, substr(hire_date, -4) year
from employees;

--이름, 전화번호, 전화번호 마지막 4자리를 조회
select first_name, phone_number, substr(phone_number, -4) back
from employees;

--모든 사원의 사번, 이름, 가린이름 순으로 출력
(이름의 앞 세굴자만 표기 뒷부분은 *로 변경)
select employee_id, first_name,
rpad(substr(first_name, 1, 3), 5, '*') result
from employees;

-replace() : 원하는 문자를 교체하는 함수
select replace('good morning', 'morning', 'night') from dual;

-length() : 문자열의 길이를 가져오는 함수
select length('hi') from dual;

--이름의 길이가 6을 초과하는 사원들만 사번, 이름을 출력
select employee_id, first_name
from employees
where length(first_name) > 6;

--전화번호의 길이가 13자 이상인 사원들의 이름, 전화번호를 조회
select first_name, phone_number
from employees
where length(phone_number) >= 13;

-concat() : 문자열을 연결하는 함수
select concat('hello', 'hi') from dual;

-|| : 문자열을 연결하는 연산자
select 'hello' || 'hi' from dual;

--사원의 이름과 성을 +로 이어붙여서 조회
select first_name || '+' || last_name fullname
from employees;
또는
select concat(concat(first_name, '+'), last_name) fullname
from employees;

-instr() : 특정 문자열의 위치를 찾아 반환하는 함수
select first_name, instr(first_name, 'ar') inx
from employees;

--이름의 마지막 2글자를 **로 대체하여 이름, 가린이름 순으로 출력
select first_name, rpad(substr(first_name, length(first_name)-2),length(first_name) , '*') hide
from employees;

--이름의 앞 한글자를 제외한 나머지 글자를 *로 대체하여 이름, 가린이름 순으로 출력
select first_name, rpad(substr(first_name,1,1),length(first_name),'*') hide
from employees;
