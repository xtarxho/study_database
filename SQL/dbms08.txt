1. 그룹함수
-avg() : 평균을 구하는 함수
-count() : 반환된 행의 갯수를 구하는 함수(null값은 무시한다)
-max() : 최대값을 구하는 함수
-min() : 최소값을 구하는 함수
-sum() : 합을 구하는 함수
# 그룹함수와 일반컬럼은 함께 사용할 수 없다.
select count(*), salary from employees

--모든 사원의 수를 구하시오
select count(*) from employees;
select count(employee_id) from employees;

--부서에 속해있는 사원의 수
select count(department_id) from employees;

--보너스를 받고있는 사원의 수
select count(commission_pct) from employees;

--선임이 있는 사원의 수
select count(manager_id) from employees;

--누군가의 상사로 일하고 있는 사원의 수
select count(distinct manager_id) from employees;

--80번 부서에 속하는 사원들의 월급의 평균을 출력
select avg(salary) from employees
where department_id = 80;

--50번 부서 사원들의 최대월급, 최소월급을 조회
select max(salary), min(salary) from employees
where department_id = 50;

--50번 부서 사원들의 월급의 합을 출력
select sum(salary) from employees
where department_id = 50;

--it_prog의 수를 출력
select count(*) from employees
where job_id = 'IT_PROG';

--sa_rep직종의 사원들의 월급의 평균, 최고월급, 최저월급, 월급의 합을 출력
select avg(salary), max(salary), min(salary), sum(salary)
from employees
where job_id = 'SA_REP';

--보너스를 받지 않는 사원의 수 출력
select count(*) from employees
where commission_pct is null;

--월급이 5000이하인 사원들의 평균 월급을 조회
select avg(salary) from employees
where salary <= 5000;

--2005년 이후 입사자의 수를 조회
select count(*) from employees
where hire_date >= '01/01/2005';

--우리회사에 있는 부서 수를 조회
select count(distinct department_id) from employees;

* nvl() : null값을 원하는 데이터로 교체하는 함수
--보너스를 받지 않는 모든 사원들의 보너스를 0.1%로 조정
select commission_pct,
	nvl(commission_pct, 0.1)
from employees

--우리회사에 있는 부서 수를 조회(대장님 포함)
select count(distinct nvl(department_id, 0)) from employees;

* dual(가상테이블) : 함수나 계산식을 테이블의 생성 없이 수행하여 볼 수 있도록 하는 가상(임시)의 테이블

2. 숫자함수
-round() : 반올림을 해 주는 함수
select round(0.123),round(0.567) from dual;
select round(0.125678, 2) from dual;

--80번 부서의 월급의 평균을 소수점 첫째자리까지 반올림
select round(avg(salary), 1) from employees
where department_id = 80;

--it_prog인 사원들의 평균 월굽을 소수점 두자리에서 반올림
select round(avg(salary),2) from employees
where job_id = 'IT_PROG';

-trunc() : 잘라내는(버림) 함수
--이름이 A로 시작하는 사원들의 평균월급을 소수점 1자리에서 절삭해서 조회
select trunc(avg(salary), 1) from employees
where first_name like 'A%';

-power() : 제곱을 구하는 함수
select power(4, 2) from dual;

-mod() : 나머지를 구하는 함수
select mod(7, 4) from dual;

--짝수 사번을 가진 사원들의 사번, 이름, 급여를 출력
select employee_id, first_name, salary from employees
where mod(employee_id, 2) = 0;

3. 문자함수
-initcap() : 첫 문자를 대문자로 변환하는 함수
-공백이나 /를 구분자로 인식한다.
select initcap('good morning') from dual;
select initcap('good/morning') from dual;

-lower() : 대문자 -> 소문자
-upper() : 소문자 -> 대문자
select lower('Good Morning') from dual;
select upper('good morning') from dual;

---밑에 두문제 다시 풀어보기---
--michael로 조회했을 때, 이름이 Michael인 사원의 사번, 이름을 조회
select employee_id, first_name
from employees
where lower(first_name) = 'michael';

--이름의 어디든 m또는 M이 들어있는 
사원의 사번, 이름을 조회
select employee_id, first_name
from employees
where lower(first_name) like '%m%';
